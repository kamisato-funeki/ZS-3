cmake_minimum_required(VERSION 3.21)
project(placement VERSION 1.0.0 LANGUAGES CXX C)

# GUI 可关闭；关闭路径不发现或链接任何 Qt 模块。
option(BUILD_GUI "Build Qt Widgets desktop application" ON)
option(BUILD_TESTING "Build GoogleTest and Qt Test suites" ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
# 用户可显式指定系统包的本地解压前缀，保持源码与机器路径分离。
set(PLACEMENT_DEPS_PREFIX "" CACHE PATH "Locally extracted development-package prefix")
if(PLACEMENT_DEPS_PREFIX)
  list(PREPEND CMAKE_PREFIX_PATH "${PLACEMENT_DEPS_PREFIX}")
  include_directories(SYSTEM "${PLACEMENT_DEPS_PREFIX}/include")
endif()
find_package(Threads REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(nlohmann_json 3.11 REQUIRED)
find_path(FFTW_INCLUDE_DIR fftw3.h HINTS "${PLACEMENT_DEPS_PREFIX}/include" REQUIRED)
find_library(FFTW_LIBRARY NAMES fftw3 libfftw3.so.3 HINTS "${PLACEMENT_DEPS_PREFIX}/lib/x86_64-linux-gnu" REQUIRED)
find_path(CIMG_INCLUDE_DIR CImg.h HINTS "${PLACEMENT_DEPS_PREFIX}/include" REQUIRED)
include(cmake/Limbo.cmake)

add_library(placement_core src/core/model.cpp)
target_include_directories(placement_core PUBLIC include)
target_link_libraries(placement_core PUBLIC Threads::Threads)
target_compile_options(placement_core PRIVATE -Wall -Wextra -Wpedantic)

# 各源码目录只属于一个目标；目录中的新实现自动触发重新配置。
foreach(module analysis io visualization application algorithms)
  if(module STREQUAL "algorithms")
    file(GLOB_RECURSE module_sources CONFIGURE_DEPENDS src/global/*.cpp src/legalization/*.cpp)
  else()
    file(GLOB_RECURSE module_sources CONFIGURE_DEPENDS "src/${module}/*.cpp")
  endif()
  if(module_sources)
    add_library(placement_${module} ${module_sources})
    target_link_libraries(placement_${module} PUBLIC placement_core)
    target_compile_options(placement_${module} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()
if(TARGET placement_io)
  target_link_libraries(placement_io PRIVATE nlohmann_json::nlohmann_json OpenSSL::Crypto limbo_lef limbo_def)
  target_link_libraries(placement_io PUBLIC placement_analysis)
endif()
if(TARGET placement_algorithms)
  target_include_directories(placement_algorithms PRIVATE "${FFTW_INCLUDE_DIR}")
  target_link_libraries(placement_algorithms PRIVATE "${FFTW_LIBRARY}")
  if(TARGET placement_analysis)
    target_link_libraries(placement_algorithms PUBLIC placement_analysis)
  endif()
endif()
if(TARGET placement_visualization)
  target_include_directories(placement_visualization SYSTEM PRIVATE "${CIMG_INCLUDE_DIR}")
  target_compile_definitions(placement_visualization PRIVATE cimg_display=0)
  target_link_libraries(placement_visualization PRIVATE limbo_gds nlohmann_json::nlohmann_json)
  target_link_libraries(placement_visualization PUBLIC placement_analysis placement_io)
endif()
if(TARGET placement_application)
  execute_process(COMMAND git -C "${CMAKE_SOURCE_DIR}" rev-parse HEAD OUTPUT_VARIABLE PLACEMENT_SOURCE_COMMIT OUTPUT_STRIP_TRAILING_WHITESPACE)
  execute_process(COMMAND git -C "${CMAKE_SOURCE_DIR}" status --porcelain OUTPUT_VARIABLE placement_dirty_output)
  set(PLACEMENT_SOURCE_DIRTY false)
  if(placement_dirty_output)
    set(PLACEMENT_SOURCE_DIRTY true)
  endif()
  file(READ "${CMAKE_SOURCE_DIR}/third_party/dependencies.lock.json" PLACEMENT_LOCK_CONTENT)
  configure_file(cmake/build_info.hpp.in generated/build_info.hpp @ONLY)
  target_include_directories(placement_application PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
  target_link_libraries(placement_application PUBLIC placement_algorithms placement_io placement_analysis placement_visualization)
  target_link_libraries(placement_application PRIVATE nlohmann_json::nlohmann_json)
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/cli/main.cpp")
  add_executable(placement_cli src/cli/main.cpp)
  target_link_libraries(placement_cli PRIVATE placement_application)
  target_compile_options(placement_cli PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(BUILD_GUI)
  find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets)
  if(BUILD_TESTING)
    find_package(Qt6 6.5 REQUIRED COMPONENTS Test)
  endif()
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/gui/main.cpp")
    file(GLOB_RECURSE gui_sources CONFIGURE_DEPENDS src/gui/*.cpp src/gui/*.hpp)
    list(REMOVE_ITEM gui_sources "${CMAKE_CURRENT_SOURCE_DIR}/src/gui/main.cpp")
    add_library(placement_gui_support ${gui_sources})
    set_target_properties(placement_gui_support PROPERTIES AUTOMOC ON)
    target_include_directories(placement_gui_support PUBLIC src/gui)
    target_link_libraries(placement_gui_support PUBLIC placement_application Qt6::Widgets)
    target_compile_options(placement_gui_support PRIVATE -Wall -Wextra -Wpedantic)
    add_executable(placement_gui src/gui/main.cpp)
    target_link_libraries(placement_gui PRIVATE placement_gui_support)
    add_executable(placement_benchmark tests/performance/benchmark.cpp)
    target_link_libraries(placement_benchmark PRIVATE placement_gui_support nlohmann_json::nlohmann_json)
  endif()
endif()

if(BUILD_TESTING)
  enable_testing()
  add_executable(placement_profile tests/performance/profile.cpp)
  target_link_libraries(placement_profile PRIVATE placement_application nlohmann_json::nlohmann_json)
  find_package(GTest REQUIRED)
  include(GoogleTest)
  file(GLOB test_sources CONFIGURE_DEPENDS tests/unit/*_test.cpp tests/integration/*_test.cpp)
  foreach(test_source IN LISTS test_sources)
    get_filename_component(test_name "${test_source}" NAME_WE)
    add_executable(${test_name} "${test_source}")
    target_link_libraries(${test_name} PRIVATE placement_core GTest::gtest_main)
    foreach(module analysis io algorithms visualization application)
      if(TARGET placement_${module})
        target_link_libraries(${test_name} PRIVATE placement_${module})
      endif()
    endforeach()
    target_compile_definitions(${test_name} PRIVATE PLACEMENT_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
    gtest_discover_tests(${test_name} DISCOVERY_TIMEOUT 60)
  endforeach()
  if(BUILD_GUI AND TARGET placement_gui_support)
    file(GLOB gui_tests CONFIGURE_DEPENDS tests/gui/*_test.cpp)
    foreach(test_source IN LISTS gui_tests)
      get_filename_component(test_name "${test_source}" NAME_WE)
      add_executable(${test_name} "${test_source}")
      set_target_properties(${test_name} PROPERTIES AUTOMOC ON)
      target_link_libraries(${test_name} PRIVATE placement_gui_support Qt6::Test)
      target_compile_definitions(${test_name} PRIVATE PLACEMENT_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
      add_test(NAME ${test_name} COMMAND ${test_name} -platform offscreen)
    endforeach()
  endif()
  if(TARGET placement_cli)
    # 独立外部回读工具可缺失，但缺失必须在验收报告中标明，不伪报该用例通过。
    find_program(KLAYOUT_EXECUTABLE klayout HINTS "${PLACEMENT_DEPS_PREFIX}/lib/klayout")
    if(PLACEMENT_DEPS_PREFIX AND EXISTS "${PLACEMENT_DEPS_PREFIX}/lib/klayout/klayout")
      set(KLAYOUT_EXECUTABLE "${PLACEMENT_DEPS_PREFIX}/lib/klayout/klayout")
    endif()
    file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/integration")
    add_test(NAME cli_headless_bmp COMMAND "${CMAKE_COMMAND}" -E env --unset=DISPLAY $<TARGET_FILE:placement_cli> export
      --aux "${CMAKE_SOURCE_DIR}/tests/fixtures/tiny/tiny.aux" --format bmp --output "${CMAKE_BINARY_DIR}/integration/headless.bmp" --overwrite)
    add_test(NAME cli_exit_contract COMMAND "${CMAKE_COMMAND}" "-DCLI=$<TARGET_FILE:placement_cli>"
      "-DSOURCE=${CMAKE_SOURCE_DIR}" "-DROOT=${CMAKE_BINARY_DIR}/integration" -P "${CMAKE_SOURCE_DIR}/tests/integration/cli_exit.cmake")
    if(KLAYOUT_EXECUTABLE)
      add_test(NAME cli_gds_fixture COMMAND placement_cli export --aux "${CMAKE_SOURCE_DIR}/tests/fixtures/tiny/tiny.aux"
        --format gds --output "${CMAKE_BINARY_DIR}/integration/layout.gds" --overwrite)
      set_tests_properties(cli_gds_fixture PROPERTIES FIXTURES_SETUP gds_fixture)
      add_test(NAME klayout_independent_readback COMMAND "${CMAKE_COMMAND}" -E env
        "LD_LIBRARY_PATH=${PLACEMENT_DEPS_PREFIX}/lib/klayout:${PLACEMENT_DEPS_PREFIX}/lib/x86_64-linux-gnu:$ENV{LD_LIBRARY_PATH}"
        "${KLAYOUT_EXECUTABLE}" -b -r "${CMAKE_SOURCE_DIR}/tests/integration/klayout_verify.py"
        -rd "gds_path=${CMAKE_BINARY_DIR}/integration/layout.gds" -rd "output_path=${CMAKE_BINARY_DIR}/integration/klayout-readback.json")
      set_tests_properties(klayout_independent_readback PROPERTIES FIXTURES_REQUIRED gds_fixture)
    endif()
  endif()
endif()
